```ts
/** 统计读接口：管理员全局事件统计（REQ-033 收尾） */
export function getTrackingSummary(days: number): {
  total: number
  byName: Array<{ name: string; count: number }>
  byDay: Array<{ day: string; count: number }>
  funnel: Array<{ name: string; count: number }>
} {
  const since = `datetime('now', ?)`
  const sinceArg = `-${days} days`
  const total = (db.prepare(`SELECT COUNT(*) AS c FROM events WHERE created_at >= ${since}`).get(sinceArg) as { c: number }).c
  const byName = db.prepare(`
    SELECT name, COUNT(*) AS count FROM events
    WHERE created_at >= ${since}
    GROUP BY name ORDER BY count DESC
  `).all(sinceArg) as Array<{ name: string; count: number }>
  const byDay = db.prepare(`
    SELECT date(created_at, 'localtime') AS day, COUNT(*) AS count FROM events
    WHERE created_at >= ${since}
    GROUP BY day ORDER BY day ASC
  `).all(sinceArg) as Array<{ day: string; count: number }>
  // 下单漏斗：按事件名顺序取总数（简单版，不去重）
  const funnelNames = [
    'order_form_start', 'order_form_step_view', 'order_form_submit_attempt',
    'order_form_submit_success', 'order_submit_success'
  ]
  const funnelRows = db.prepare(`
    SELECT name, COUNT(*) AS count FROM events
    WHERE created_at >= ${since} AND name IN (${funnelNames.map(() => '?').join(',')})
    GROUP BY name
  `).all(sinceArg, ...funnelNames) as Array<{ name: string; count: number }>
  const countMap = new Map(funnelRows.map(r => [r.name, r.count]))
  const funnel = funnelNames.map(name => ({ name, count: countMap.get(name) ?? 0 }))
  return { total, byName, byDay, funnel }
}

/** 画师自己的统计（artist_id 过滤） */
export function getArtistTrackingSummary(artistId: number, days: number): {
  total: number
  byName: Array<{ name: string; count: number }>
} {
  const since = `datetime('now', ?)`
  const sinceArg = `-${days} days`
  const total = (db.prepare(`
    SELECT COUNT(*) AS c FROM events
    WHERE artist_id = ? AND created_at >= ${since}
  `).get(artistId, sinceArg) as { c: number }).c
  const byName = db.prepare(`
    SELECT name, COUNT(*) AS count FROM events
    WHERE artist_id = ? AND created_at >= ${since}
    GROUP BY name ORDER BY count DESC
  `).all(artistId, sinceArg) as Array<{ name: string; count: number }>
  return { total, byName }
}

/** 管理员开关：画师门面统计显隐（platform_config key=artist_stats_visible，默认 true） */
export function getArtistStatsVisible(): boolean {
  const row = db.prepare(`SELECT value FROM platform_config WHERE key = 'artist_stats_visible'`).get() as { value: string } | undefined
  return row ? row.value !== 'false' : true
}
export function setArtistStatsVisible(visible: boolean): boolean {
  db.prepare(`
    INSERT INTO platform_config (key, value) VALUES ('artist_stats_visible', ?)
    ON CONFLICT(key) DO UPDATE SET value = excluded.value
  `).run(visible ? 'true' : 'false')
  return visible
}
```

> ⚠️ SQL 写法注意：上例用模板字符串拼 `${since}` 是**占位符复用技巧**（同一占位符可用多次），`sinceArg` 作为参数传入。也可直接写 `created_at >= datetime('now', '-30 days')` 字面量（days 已由路由层 clamp，无注入面），**推荐字面量写法更简单**：
> `const since = `datetime('now', '-${days} days')`` 然后 `WHERE created_at >= ${since}`——days 必须先在路由层 `Math.min(Math.max(parseInt(days,10)||30,1),90)` 整形，保证无注入。二选一，确保测试通过即可。

```ts
  /** GET /api/admin/tracking/summary — 管理员全局事件统计（REQ-033 收尾） */
  fastify.get('/api/admin/tracking/summary', { preHandler: requireAdmin }, async (request) => {
    const { days } = request.query as { days?: string }
    const d = Math.min(Math.max(parseInt(days, 10) || 30, 1), 90)
    return trackingService.getTrackingSummary(d)
  })

  /** GET /api/artist/tracking/summary — 画师自己的事件统计（门面区块） */
  fastify.get('/api/artist/tracking/summary', { preHandler: requireAuth }, async (request) => {
    const artist = (request as unknown as { artist?: Artist }).artist as Artist
    const { days } = request.query as { days?: string }
    const d = Math.min(Math.max(parseInt(days, 10) || 14, 1), 90)
    const summary = trackingService.getArtistTrackingSummary(artist.id, d)
    return { enabled: trackingService.getArtistStatsVisible(), ...summary }
  })

  /** GET /api/admin/tracking-config — 读管理员开关 */
  fastify.get('/api/admin/tracking-config', { preHandler: requireAdmin }, async () => {
    return { artistStatsVisible: trackingService.getArtistStatsVisible() }
  })

  /** PUT /api/admin/tracking-config — 写管理员开关 */
  fastify.put('/api/admin/tracking-config', {
    preHandler: requireAdmin,
    schema: { body: { type: 'object', required: ['artistStatsVisible'], properties: { artistStatsVisible: { type: 'boolean' } }, additionalProperties: false } }
  }, async (request) => {
    const { artistStatsVisible } = request.body as { artistStatsVisible: boolean }
    return { artistStatsVisible: trackingService.setArtistStatsVisible(artistStatsVisible) }
  })
```
